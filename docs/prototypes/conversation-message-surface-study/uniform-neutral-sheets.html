下面是启动过程必须遵守的核心判断。它只读取已经保存的 delivery，不在每次启动时重新生成默认值。

// v74 owns the one-time backfill boundary.
pub fn reconcile_skill_delivery(
  skill: &ManagedSkill,
  stored_groups: &[RuntimeGroup],
  migration_version: i64,
) -> DeliveryPlan {
  if migration_version == 74 && stored_groups.is_empty() {
    return DeliveryPlan::all_current_groups(
      skill.enabled,
      "migration-v74",
    );
  }

  DeliveryPlan::preserve(
    skill.enabled,
    stored_groups,
    "user-owned-selection",
  )
}

// Startup must never repopulate a group removed after migration.
debug_assert!(plan.source != "startup-default");
